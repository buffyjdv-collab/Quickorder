   184→// ─── Orders Section ───────────────────────────────────────────────────
   185→function OrdersSection({ orders }: { orders: Order[] }) {
   186→  const [search, setSearch] = React.useState('');
   187→  const [activeOnly, setActiveOnly] = React.useState(false);
   188→  const filtered = React.useMemo(() => {
   189→    const q = search.trim().toLowerCase();
   190→    return orders.filter(o => {
   191→      if (activeOnly && (o.status === 'completed' || o.status === 'cancelled')) return false;
   192→      if (!q) return true;
   193→      return String(o.orderNumber).includes(q) || (o.tableCode || '').toLowerCase().includes(q) || (o.customerName || '').toLowerCase().includes(q);
   194→    });
   195→  }, [orders, search, activeOnly]);
   196→
   197→  const byStatus = React.useMemo(() => {
   198→    const map: Record<string, Order[]> = {};
   199→    for (const s of ORDER_STATUS_FLOW) map[s] = [];
   200→    for (const o of filtered) { if (map[o.status]) map[o.status].push(o); }
   201→    return map;
   202→  }, [filtered]);
   203→
   204→  return (
   205→    <div className='space-y-3'>
   206→      {/* Search bar */}
   207→      <div className='flex flex-wrap items-center gap-2'>
   208→        <div className='relative flex-1 min-w-[180px]'>
   209→          <Search className='absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground' />
   210→          <Input placeholder='Search order #, table…' value={search} onChange={e => setSearch(e.target.value)} className='pl-8 h-9 text-sm' />
   211→        </div>
   212→        <Badge variant={activeOnly ? 'default' : 'secondary'} className='cursor-pointer h-8 gap-1.5 px-3 select-none' onClick={() => setActiveOnly(!activeOnly)}>
   213→          {activeOnly ? 'All' : 'Active'} · {filtered.length}
   214→        </Badge>
   215→      </div>
   216→
   217→      {/* Desktop: Kanban columns */}
   218→      <div className='hidden lg:block'>
   219→        <div className='flex gap-3 overflow-x-auto pb-2'>
   220→          {ORDER_STATUS_FLOW.map(status => {
   221→            const meta = ORDER_STATUS_META[status];
   222→            const cols = byStatus[status] || [];
   223→            return (
   224→              <div key={status} className='flex min-w-[260px] flex-1 flex-col gap-2'>
   225→                <div className={cn('sticky top-0 z-10 flex items-center justify-between rounded-lg px-3 py-2 ring-1', meta.bg, meta.ring)}>
   226→                  <span className='flex items-center gap-2 text-sm font-semibold'><span>{meta.emoji}</span>{meta.label}</span>
   227→                  <Badge variant='secondary' className={cn('bg-white/70 tabular-nums', meta.color)}>{cols.length}</Badge>
   228→                </div>
   229→                <ScrollArea className='max-h-[calc(100vh-280px)]'>
   230→                  <div className='space-y-2 pr-1'>
   231→                    {cols.length === 0 ? <div className='flex h-20 items-center justify-center rounded-lg border border-dashed text-xs text-muted-foreground'>No orders</div> : cols.map(o => <OrderCard key={o.id} order={o} />)}
   232→                  </div>
   233→                </ScrollArea>
   234→              </div>);
   235→          })}
   236→        </div>
   237→      </div>
   238→
   239→      {/* Mobile: List view grouped by status */}
   240→      <div className='lg:hidden space-y-4'>
   241→        {ORDER_STATUS_FLOW.map(status => {
   242→          const meta = ORDER_STATUS_META[status];
   243→          const cols = byStatus[status] || [];
   244→          if (cols.length === 0) return null;
   245→          return (
   246→            <div key={status}>
   247→              <div className={cn('flex items-center justify-between rounded-lg px-3 py-2 mb-2 ring-1', meta.bg, meta.ring)}>
   248→                <span className='flex items-center gap-2 text-sm font-semibold'><span>{meta.emoji}</span>{meta.label}</span>
   249→                <Badge variant='secondary' className={cn('bg-white/70 tabular-nums', meta.color)}>{cols.length}</Badge>
   250→              </div>
   251→              <div className='space-y-2'>{cols.map(o => <OrderCard key={o.id} order={o} compact />)}</div>
   252→            </div>);
   253→        })}
   254→      </div>
   255→    </div>
   256→  );
   257→}

   263→    <Card className={cn('shadow-sm transition-shadow hover:shadow-md border-l-4', meta.ring.replace('ring-', 'border-l-'))}>
   264→      <CardContent className='p-3 space-y-2'>

   267→            <div className='flex flex-wrap items-center gap-1.5'>
   268→              <span className='font-mono text-sm font-bold'>#{order.orderNumber}</span>
   269→              {order.tableCode ? <Badge variant='secondary' className='bg-emerald-50 text-emerald-700 text-[10px]'>🪑 {order.tableCode}</Badge> : <Badge variant='secondary' className='text-[10px]'>🥡 Counter</Badge>}
   270→              <Badge className={cn('text-[10px]', meta.bg, meta.color)}>{meta.label}</Badge>
   271→            </div>
   272→            <p className='mt-0.5 flex items-center gap-1 text-[11px] text-muted-foreground'><Clock className='size-3' />{formatTimeAgo(order.createdAt)}</p>
   273→          </div>
   274→          <div className='text-right shrink-0'>
   275→            <p className='font-mono text-sm font-bold tabular-nums'>₹{order.total.toLocaleString()}</p>
   276→            <p className='text-[10px] text-muted-foreground'>{payMeta.emoji} {payMeta.label}</p>
   277→          </div>
   278→        </div>
   279→        {!compact && order.items.length > 0 && (
   280→          <div className='max-h-[120px] space-y-1 overflow-y-auto rounded-md bg-muted/30 p-2'>
   281→            {order.items.map(it => (
   282→              <div key={it.id} className='flex items-center gap-2 text-xs'>
   283→                <span>{it.productEmoji}</span>
   284→                <span className='flex-1 truncate'>{it.productName}</span>

   286→                <span className='font-mono tabular-nums'>₹{it.totalPrice}</span>
   287→              </div>
   288→            ))}
   289→          </div>
   290→        )}
   291→      </CardContent>
   292→    </Card>
   293→  );
   294→}
