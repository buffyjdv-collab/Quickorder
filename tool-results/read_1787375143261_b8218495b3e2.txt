   185→// ─── Orders Section ───────────────────────────────────────────────────
   186→function OrdersSection({ orders }: { orders: Order[] }) {
   187→  const [search, setSearch] = React.useState('');
   188→  const [activeOnly, setActiveOnly] = React.useState(false);
   189→  const filtered = React.useMemo(() => {
   190→    const q = search.trim().toLowerCase();
   191→    return orders.filter(o => {
   192→      if (activeOnly && (o.status === 'completed' || o.status === 'cancelled')) return false;
   193→      if (!q) return true;
   194→      return String(o.orderNumber).includes(q) || (o.tableCode || '').toLowerCase().includes(q) || (o.customerName || '').toLowerCase().includes(q);
   195→    });
   196→  }, [orders, search, activeOnly]);
   197→
   198→  const byStatus = React.useMemo(() => {
   199→    const map: Record<string, Order[]> = {};
   200→    for (const s of ORDER_STATUS_FLOW) map[s] = [];
   201→    for (const o of filtered) { if (map[o.status]) map[o.status].push(o); }
   202→    return map;
   203→  }, [filtered]);
   204→
   205→  return (
   206→    <div className='space-y-3'>
   207→      {/* Search bar */}
   208→      <div className='flex flex-wrap items-center gap-2'>
   209→        <div className='relative flex-1 min-w-[180px]'>
   210→          <Search className='absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground' />
   211→          <Input placeholder='Search order #, table…' value={search} onChange={e => setSearch(e.target.value)} className='pl-8 h-9 text-sm' />
   212→        </div>
   213→        <Badge variant={activeOnly ? 'default' : 'secondary'} className='cursor-pointer h-8 gap-1.5 px-3 select-none' onClick={() => setActiveOnly(!activeOnly)}>
   214→          {activeOnly ? 'All' : 'Active'} · {filtered.length}
   215→        </Badge>
   216→      </div>
   217→
   218→      {/* Desktop: Kanban columns */}
   219→      <div className='hidden lg:block'>
   220→        <div className='flex gap-3 overflow-x-auto pb-2'>
   221→          {ORDER_STATUS_FLOW.map(status => {
   222→            const meta = ORDER_STATUS_META[status];
   223→            const cols = byStatus[status] || [];
   224→            return (
   225→              <div key={status} className='flex min-w-[260px] flex-1 flex-col gap-2'>
   226→                <div className={cn('sticky top-0 z-10 flex items-center justify-between rounded-lg px-3 py-2 ring-1', meta.bg, meta.ring)}>
   227→                  <span className='flex items-center gap-2 text-sm font-semibold'><span>{meta.emoji}</span>{meta.label}</span>
   228→                  <Badge variant='secondary' className={cn('bg-white/70 tabular-nums', meta.color)}>{cols.length}</Badge>
   229→                </div>
   230→                <ScrollArea className='max-h-[calc(100vh-280px)]'>
   231→                  <div className='space-y-2 pr-1'>
   232→                    {cols.length === 0 ? <div className='flex h-20 items-center justify-center rounded-lg border border-dashed text-xs text-muted-foreground'>No orders</div> : cols.map(o => <OrderCard key={o.id} order={o} />)}
   233→                  </div>
   234→                </ScrollArea>
   235→              </div>);
   236→          })}
   237→        </div>
   238→      </div>
   239→
   240→      {/* Mobile: List view grouped by status */}
   241→      <div className='lg:hidden space-y-4'>
   242→        {ORDER_STATUS_FLOW.map(status => {
   243→          const meta = ORDER_STATUS_META[status];
   244→          const cols = byStatus[status] || [];
   245→          if (cols.length === 0) return null;
   246→          return (
   247→            <div key={status}>
   248→              <div className={cn('flex items-center justify-between rounded-lg px-3 py-2 mb-2 ring-1', meta.bg, meta.ring)}>
   249→                <span className='flex items-center gap-2 text-sm font-semibold'><span>{meta.emoji}</span>{meta.label}</span>
   250→                <Badge variant='secondary' className={cn('bg-white/70 tabular-nums', meta.color)}>{cols.length}</Badge>
   251→              </div>
   252→              <div className='space-y-2'>{cols.map(o => <OrderCard key={o.id} order={o} compact />)}</div>
   253→            </div>);
   254→        })}
   255→      </div>
   256→    </div>
   257→  );
   258→}

   264→    <Card className={cn('shadow-sm transition-shadow hover:shadow-md border-l-4', meta.ring.replace('ring-', 'border-l-'))}>
   265→      <CardContent className='p-3 space-y-2'>

   268→            <div className='flex flex-wrap items-center gap-1.5'>
   269→              <span className='font-mono text-sm font-bold'>#{order.orderNumber}</span>
   270→              {order.tableCode ? <Badge variant='secondary' className='bg-emerald-50 text-emerald-700 text-[10px]'>🪑 {order.tableCode}</Badge> : <Badge variant='secondary' className='text-[10px]'>🥡 Counter</Badge>}
   271→              <Badge className={cn('text-[10px]', meta.bg, meta.color)}>{meta.label}</Badge>
   272→            </div>
   273→            <p className='mt-0.5 flex items-center gap-1 text-[11px] text-muted-foreground'><Clock className='size-3' />{formatTimeAgo(order.createdAt)}</p>
   274→          </div>
   275→          <div className='text-right shrink-0'>
   276→            <p className='font-mono text-sm font-bold tabular-nums'>₹{order.total.toLocaleString()}</p>

   278→          </div>
   279→        </div>
   280→        {!compact && order.items.length > 0 && (
   281→          <div className='max-h-[120px] space-y-1 overflow-y-auto rounded-md bg-muted/30 p-2'>
   282→            {order.items.map(it => (
   283→              <div key={it.id} className='flex items-center gap-2 text-xs'>

   287→                <span className='font-mono tabular-nums'>₹{it.totalPrice}</span>
   288→              </div>
   289→            ))}
   290→          </div>
   291→        )}
   292→      </CardContent>
   293→    </Card>
   294→  );
   295→}
